cmake_minimum_required(VERSION 3.20)
project(GameBoyStudio VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
# The native Studio hardware implementation is header-only and built into every
# consumer. No external emulator library or prebuilt engine DLL is required.
add_library(gb_studio_core INTERFACE)
target_include_directories(gb_studio_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(gb_studio_core INTERFACE cxx_std_20)
find_package(Threads REQUIRED)
if(WIN32)
    # Windows macros must not replace std::min/std::max in tests or shared headers.
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
# Keep native hardware module dependencies explicit for localized MSVC /showIncludes.
set_source_files_properties(src/headless.cpp src/windows_main.cpp
    tests/core_tests.cpp tests/cgb_memory_tests.cpp tests/cgb_ppu_tests.cpp
    tests/cgb_apu_tests.cpp tests/cgb_integration_tests.cpp
    tests/core_extension_tests.cpp tests/session_tests.cpp tests/studio_tools_tests.cpp
    tests/audio_buffer_tests.cpp tests/cartridge_tools_tests.cpp PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/gameboy.hpp;${CMAKE_CURRENT_SOURCE_DIR}/src/native_ppu.hpp;${CMAKE_CURRENT_SOURCE_DIR}/src/native_apu.hpp;${CMAKE_CURRENT_SOURCE_DIR}/src/native_noise.hpp;${CMAKE_CURRENT_SOURCE_DIR}/src/state_codec.hpp")
add_library(gb_miniz STATIC third_party/miniz/miniz.c third_party/miniz/miniz_tdef.c third_party/miniz/miniz_tinfl.c third_party/miniz/miniz_zip.c)
target_include_directories(gb_miniz PUBLIC third_party/miniz)
file(GLOB GB_LUA_SOURCES CONFIGURE_DEPENDS third_party/lua-src/l*.c)
list(FILTER GB_LUA_SOURCES EXCLUDE REGEX "/(lua|luac|ltests)\\.c$")
add_library(gb_lua STATIC ${GB_LUA_SOURCES})
target_include_directories(gb_lua PUBLIC third_party/lua-src)
if(NOT WIN32)
    target_link_libraries(gb_lua PUBLIC m)
endif()
file(GLOB_RECURSE GB_ACHIEVEMENT_SOURCES CONFIGURE_DEPENDS third_party/rcheevos/src/api/*.c third_party/rcheevos/src/client/*.c third_party/rcheevos/src/hash/*.c third_party/rcheevos/src/runtime/*.c third_party/rcheevos/src/util/*.c)
add_library(gb_achievements STATIC ${GB_ACHIEVEMENT_SOURCES})
target_include_directories(gb_achievements PUBLIC third_party/rcheevos/include)
target_compile_definitions(gb_achievements PUBLIC RC_CLIENT_SUPPORTS_HASH)
if(NOT WIN32)
    target_link_libraries(gb_achievements PUBLIC m)
endif()
add_executable(gb_pixel_filters src/pixel_filter_main.cpp third_party/xbrz/xbrz.cpp)
option(GB_BUILD_COVERAGE_FUZZER "Build GCC PC-feedback native Studio loader fuzzer (Linux)" OFF)
if(GB_BUILD_COVERAGE_FUZZER)
    if(WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        message(FATAL_ERROR "The PC-feedback fuzzer requires a native Linux GCC build.")
    endif()
    add_library(gb_cov_trace STATIC tests/coverage_trace.c)
    target_compile_options(gb_cov_trace PRIVATE -fno-sanitize-coverage=trace-pc)
    add_executable(gb_coverage_fuzzer tests/coverage_fuzzer.cpp)
    target_compile_options(gb_coverage_fuzzer PRIVATE -fsanitize-coverage=trace-pc -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(gb_coverage_fuzzer PRIVATE -fsanitize=address,undefined)
    target_link_libraries(gb_coverage_fuzzer PRIVATE gb_studio_core gb_cov_trace)
endif()
if(MINGW)
    target_link_options(gb_pixel_filters PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_executable(gb_headless src/headless.cpp)
target_link_libraries(gb_headless PRIVATE gb_studio_core)
if(WIN32)
    target_link_libraries(gb_headless PRIVATE shell32)
endif()
if(BUILD_TESTING)
    foreach(accuracy_suite cpu_dma native_apu native_ppu)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${accuracy_suite}_accuracy_tests.cpp")
            add_executable(gb_${accuracy_suite}_accuracy_tests tests/${accuracy_suite}_accuracy_tests.cpp)
            target_link_libraries(gb_${accuracy_suite}_accuracy_tests PRIVATE gb_studio_core Threads::Threads)
            add_test(NAME ${accuracy_suite}_accuracy COMMAND gb_${accuracy_suite}_accuracy_tests)
        endif()
    endforeach()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/accuracy_state_tests.cpp")
        add_executable(gb_accuracy_state_tests tests/accuracy_state_tests.cpp)
        target_link_libraries(gb_accuracy_state_tests PRIVATE gb_studio_core Threads::Threads)
        add_test(NAME accuracy_state COMMAND gb_accuracy_state_tests)
    endif()
    add_executable(gb_frame_pacing_tests tests/frame_pacing_tests.cpp)
    add_test(NAME frame_pacing COMMAND gb_frame_pacing_tests)
    add_executable(gb_native_core_regression_tests tests/native_core_regression_tests.cpp)
    add_test(NAME native_core_regressions COMMAND gb_native_core_regression_tests)
    add_executable(gb_native_state_fuzz_tests tests/native_state_fuzz_tests.cpp)
    add_test(NAME native_state_fuzz COMMAND gb_native_state_fuzz_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
    add_executable(gb_native_integration_tests tests/native_integration_tests.cpp)
    target_link_libraries(gb_native_integration_tests PRIVATE Threads::Threads)
    if(WIN32)
        target_link_libraries(gb_native_integration_tests PRIVATE ws2_32)
    endif()
    add_test(NAME native_integration COMMAND gb_native_integration_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb" "${CMAKE_CURRENT_SOURCE_DIR}/demo.gbc")
    add_executable(gb_integration_pack_tests tests/integration_pack_tests.cpp)
    target_link_libraries(gb_integration_pack_tests PRIVATE gb_lua Threads::Threads)
    add_test(NAME integration_pack COMMAND gb_integration_pack_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
    add_executable(gb_achievement_tests tests/achievement_tests.cpp)
    target_link_libraries(gb_achievement_tests PRIVATE gb_achievements Threads::Threads)
    if(WIN32)
        target_link_libraries(gb_achievement_tests PRIVATE winhttp)
    endif()
    add_test(NAME achievements COMMAND gb_achievement_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
    add_executable(gb_core_tests tests/core_tests.cpp)
    add_test(NAME core COMMAND gb_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
    foreach(suite cgb_memory cgb_ppu cgb_apu cgb_integration)
        add_executable(gb_${suite}_tests tests/${suite}_tests.cpp)
        if(suite STREQUAL "cgb_memory")
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/dmg-v0.6.state" "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
        elseif(suite STREQUAL "cgb_integration")
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gbc")
        else()
            add_test(NAME ${suite} COMMAND gb_${suite}_tests)
        endif()
    endforeach()
    foreach(suite core_extension session studio_tools audio_buffer cartridge_tools feature_pack loader_fuzz)
        add_executable(gb_${suite}_tests tests/${suite}_tests.cpp)
        target_link_libraries(gb_${suite}_tests PRIVATE Threads::Threads)
        if(suite STREQUAL "loader_fuzz")
            target_link_libraries(gb_${suite}_tests PRIVATE gb_miniz)
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
        elseif(suite STREQUAL "feature_pack")
            target_link_libraries(gb_${suite}_tests PRIVATE gb_miniz gb_lua)
            if(WIN32)
                target_link_libraries(gb_${suite}_tests PRIVATE ws2_32)
            endif()
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb" "${CMAKE_CURRENT_SOURCE_DIR}/demo.gbc")
        elseif(suite STREQUAL "studio_tools" OR suite STREQUAL "audio_buffer" OR suite STREQUAL "cartridge_tools")
            add_test(NAME ${suite} COMMAND gb_${suite}_tests)
        else()
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
        endif()
    endforeach()
endif()
option(GB_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer in tests" OFF)
foreach(target gb_cpu_dma_accuracy_tests gb_native_apu_accuracy_tests gb_native_ppu_accuracy_tests gb_accuracy_state_tests gb_frame_pacing_tests gb_native_core_regression_tests gb_native_state_fuzz_tests gb_native_integration_tests gb_integration_pack_tests gb_headless gb_core_tests gb_cgb_memory_tests gb_cgb_ppu_tests gb_cgb_apu_tests gb_cgb_integration_tests gb_core_extension_tests gb_session_tests gb_studio_tools_tests gb_audio_buffer_tests gb_cartridge_tools_tests gb_feature_pack_tests gb_loader_fuzz_tests gb_miniz gb_lua gb_achievements gb_achievement_tests)
    if(TARGET ${target})
        get_target_property(gb_target_type ${target} TYPE)
        if(gb_target_type STREQUAL "EXECUTABLE")
            target_link_libraries(${target} PRIVATE gb_studio_core)
        endif()
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
            if(NOT target STREQUAL "gb_miniz" AND NOT target STREQUAL "gb_lua")
                target_link_options(${target} PRIVATE /STACK:8388608)
            endif()
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-misleading-indentation)
            if(MINGW AND NOT target STREQUAL "gb_miniz" AND NOT target STREQUAL "gb_lua")
                target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
            endif()
            if(GB_ENABLE_SANITIZERS)
                target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
                target_link_options(${target} PRIVATE -fsanitize=address,undefined)
            endif()
        endif()
    endif()
endforeach()

    add_executable(GBStudio WIN32 src/windows_main.cpp)
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT GBStudio)
    if(MSVC)

    else()
        set(GB_CAPTURE_LIBRARY avicap32)

    target_link_libraries(GBStudio PRIVATE gb_studio_core user32 gdi32 comdlg32 shell32 winmm ole32 uuid avrt d3d11 dxgi d3dcompiler ws2_32 comctl32 windowscodecs ${GB_CAPTURE_LIBRARY} winhttp gb_miniz gb_lua gb_achievements Threads::Threads)

    if(MSVC)
        target_compile_options(GBStudio PRIVATE /W3 /EHsc /utf-8)
    else()
        target_compile_options(GBStudio PRIVATE -Wall -Wextra -Wno-misleading-indentation -Wno-unknown-pragmas)
        if(MINGW)
            target_link_options(GBStudio PRIVATE -municode -static -static-libgcc -static-libstdc++)
        endif()
    endif()
    add_custom_command(TARGET GBStudio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb" "$<TARGET_FILE_DIR:GBStudio>/demo.gb")

        "${CMAKE_CURRENT_SOURCE_DIR}/demo.gbc" "$<TARGET_FILE_DIR:GBStudio>/demo.gbc")
    install(TARGETS GBStudio gb_pixel_filters RUNTIME DESTINATION .)
    install(TARGETS gb_headless RUNTIME DESTINATION tools)
    install(FILES INSTALL_WINDOWS.bat INSTALL_WINDOWS.ps1 UNINSTALL_WINDOWS.ps1 SIGN_WINDOWS.ps1 RUN_WINDOWS_QA.ps1 DESTINATION .)
    install(FILES demo.gb demo.gbc README.md README_RU.md COMPATIBILITY.md LICENSE DESTINATION .)
    install(DIRECTORY docs DESTINATION . FILES_MATCHING PATTERN "*.md" PATTERN "*.json" PATTERN "*.tsv" PATTERN "*.txt" PATTERN "*.xml")
    install(FILES third_party/rcheevos/LICENSE DESTINATION licenses RENAME rcheevos-MIT.txt)

    install(FILES third_party/xbrz/COPYING DESTINATION licenses RENAME xBRZ-GPL3.txt)

    install(FILES third_party/hq2x/hq2x.glsl DESTINATION licenses RENAME HQ2x-MIT-source.txt)
    install(FILES third_party/lua-src/lua.h DESTINATION licenses RENAME Lua-MIT-source.txt)
    install(FILES docs/THIRD_PARTY_NOTICES.md DESTINATION licenses)
    install(FILES src/pixel_filter_main.cpp DESTINATION filter-source/src)
    install(DIRECTORY third_party/xbrz DESTINATION filter-source/third_party PATTERN ".git" EXCLUDE)
    install(FILES packaging/filter-CMakeLists.txt DESTINATION filter-source RENAME CMakeLists.txt)
    install(FILES packaging/filter-README.md DESTINATION filter-source RENAME README.md)

endif()

if(MINGW)
    # Native tests and Studio tools can hold several complete console instances.
    # Match the MSVC test reserve instead of MinGW's default 2 MiB stack.
    get_property(gb_build_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(gb_target IN LISTS gb_build_targets)
        get_target_property(gb_target_type ${gb_target} TYPE)
        if(gb_target_type STREQUAL "EXECUTABLE")
            target_link_options(${gb_target} PRIVATE "LINKER:--stack,8388608")
        endif()
    endforeach()
endif()
