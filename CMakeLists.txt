cmake_minimum_required(VERSION 3.20)
project(GameBoyStudio VERSION 0.8 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
find_package(Threads REQUIRED)
# Keep the monolithic core dependency explicit for localized MSVC /showIncludes.
set_source_files_properties(src/headless.cpp src/windows_main.cpp
    tests/core_tests.cpp tests/cgb_memory_tests.cpp tests/cgb_ppu_tests.cpp
    tests/cgb_apu_tests.cpp tests/cgb_integration_tests.cpp
    tests/core_extension_tests.cpp tests/session_tests.cpp tests/studio_tools_tests.cpp
    tests/audio_buffer_tests.cpp tests/cartridge_tools_tests.cpp PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/gameboy.hpp")
add_executable(gb_headless src/headless.cpp)
if(WIN32)
    target_link_libraries(gb_headless PRIVATE shell32)
endif()
if(BUILD_TESTING)
    add_executable(gb_core_tests tests/core_tests.cpp)
    add_test(NAME core COMMAND gb_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
    foreach(suite cgb_memory cgb_ppu cgb_apu cgb_integration)
        add_executable(gb_${suite}_tests tests/${suite}_tests.cpp)
        if(suite STREQUAL "cgb_memory")
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/dmg-v0.6.state" "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
        elseif(suite STREQUAL "cgb_integration")
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gbc")
        else()
            add_test(NAME ${suite} COMMAND gb_${suite}_tests)
        endif()
    endforeach()
    foreach(suite core_extension session studio_tools audio_buffer cartridge_tools)
        add_executable(gb_${suite}_tests tests/${suite}_tests.cpp)
        target_link_libraries(gb_${suite}_tests PRIVATE Threads::Threads)
        if(suite STREQUAL "studio_tools" OR suite STREQUAL "audio_buffer" OR suite STREQUAL "cartridge_tools")
            add_test(NAME ${suite} COMMAND gb_${suite}_tests)
        else()
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
        endif()
    endforeach()
endif()
option(GB_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer in tests" OFF)
foreach(target gb_headless gb_core_tests gb_cgb_memory_tests gb_cgb_ppu_tests gb_cgb_apu_tests gb_cgb_integration_tests gb_core_extension_tests gb_session_tests gb_studio_tools_tests gb_audio_buffer_tests gb_cartridge_tools_tests)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-misleading-indentation)
            if(GB_ENABLE_SANITIZERS)
                target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
                target_link_options(${target} PRIVATE -fsanitize=address,undefined)
            endif()
        endif()
    endif()
endforeach()
if(WIN32)
    add_executable(GBStudio WIN32 src/windows_main.cpp)
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT GBStudio)
    target_link_libraries(GBStudio PRIVATE user32 gdi32 comdlg32 shell32 winmm ole32 uuid avrt Threads::Threads)
    target_compile_definitions(GBStudio PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
        target_compile_options(GBStudio PRIVATE /W3 /EHsc /utf-8)
    else()
        target_compile_options(GBStudio PRIVATE -Wall -Wextra -Wno-misleading-indentation -Wno-unknown-pragmas)
        if(MINGW)
            target_link_options(GBStudio PRIVATE -municode -static -static-libgcc -static-libstdc++)
        endif()
    endif()
    add_custom_command(TARGET GBStudio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb" "$<TARGET_FILE_DIR:GBStudio>/demo.gb")
    add_custom_command(TARGET GBStudio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/demo.gbc" "$<TARGET_FILE_DIR:GBStudio>/demo.gbc")
    install(TARGETS GBStudio RUNTIME DESTINATION .)
    install(TARGETS gb_headless RUNTIME DESTINATION tools)
    install(FILES demo.gb demo.gbc README.md README_RU.md COMPATIBILITY.md LICENSE DESTINATION .)
    install(DIRECTORY docs DESTINATION . FILES_MATCHING PATTERN "*.md" PATTERN "*.json")
endif()
