    tests/cgb_apu_tests.cpp tests/cgb_integration_tests.cpp

# SameBoy is GNU C. Windows MSVC builds use the supplied C ABI DLL;
# MinGW/Clang can rebuild it from the fully vendored source.
if(NOT MSVC)
    file(GLOB GB_ACCURACY_SOURCES CONFIGURE_DEPENDS third_party/sameboy/Core/*.c)
    list(FILTER GB_ACCURACY_SOURCES EXCLUDE REGEX "/(debugger|sm83_disassembler|symbol_hash|rewind|cheat_search|cheats)\\.c$")
    if(WIN32)
        add_library(gb_accuracy SHARED ${GB_ACCURACY_SOURCES} src/accuracy_bridge.c)
        set_target_properties(gb_accuracy PROPERTIES PREFIX "" OUTPUT_NAME "gb_accuracy")
    else()
        add_library(gb_accuracy STATIC ${GB_ACCURACY_SOURCES} src/accuracy_bridge.c)
        target_link_libraries(gb_accuracy PUBLIC m)
    endif()
    set_target_properties(gb_accuracy PROPERTIES C_EXTENSIONS ON)
    target_compile_definitions(gb_accuracy PRIVATE GB_INTERNAL _GNU_SOURCE)
    target_compile_definitions(gb_accuracy PUBLIC GB_DISABLE_DEBUGGER GB_DISABLE_CHEAT_SEARCH GB_DISABLE_CHEATS GB_DISABLE_REWIND GB_DISABLE_TIMEKEEPING GB_VERSION="1.0.4")
    if(MINGW)
        target_link_options(gb_accuracy PRIVATE -static-libgcc)
    endif()
endif()
if(MSVC)
    foreach(configuration Debug Release RelWithDebInfo MinSizeRel)
        file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${configuration}")
        configure_file(third_party/sameboy/prebuilt/gb_accuracy.dll "${CMAKE_CURRENT_BINARY_DIR}/${configuration}/gb_accuracy.dll" COPYONLY)
    endforeach()
endif()

if(BUILD_TESTING)

        target_compile_definitions(gb_ppu_latch_state_tests PRIVATE GB_INTERNAL _GNU_SOURCE)
        set_target_properties(gb_ppu_latch_state_tests PROPERTIES C_EXTENSIONS ON)

            target_compile_definitions(gb_ppu_latch_state_tests PRIVATE GB_DISABLE_DEBUGGER GB_DISABLE_CHEAT_SEARCH GB_DISABLE_CHEATS GB_DISABLE_REWIND GB_DISABLE_TIMEKEEPING GB_VERSION="1.0.4")

        add_executable(gb_${suite}_tests tests/${suite}_tests.cpp)
        if(suite STREQUAL "cgb_memory")
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/dmg-v0.6.state" "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
        elseif(suite STREQUAL "cgb_integration")
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gbc")
        else()
            add_test(NAME ${suite} COMMAND gb_${suite}_tests)
        endif()

            target_link_libraries(gb_${suite}_tests PRIVATE gb_miniz)
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
        elseif(suite STREQUAL "feature_pack")
            target_link_libraries(gb_${suite}_tests PRIVATE gb_miniz gb_lua)
            if(WIN32)
                target_link_libraries(gb_${suite}_tests PRIVATE ws2_32)
            endif()
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb" "${CMAKE_CURRENT_SOURCE_DIR}/demo.gbc")
        elseif(suite STREQUAL "studio_tools" OR suite STREQUAL "audio_buffer" OR suite STREQUAL "cartridge_tools")
            add_test(NAME ${suite} COMMAND gb_${suite}_tests)
        else()
            add_test(NAME ${suite} COMMAND gb_${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
        endif()
    endforeach()

option(GB_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer in tests" OFF)
foreach(target gb_frame_pacing_tests gb_ppu_latch_state_tests gb_accuracy_fuzz_tests gb_integration_pack_tests gb_headless gb_core_tests gb_cgb_memory_tests gb_cgb_ppu_tests gb_cgb_apu_tests gb_cgb_integration_tests gb_core_extension_tests gb_session_tests gb_studio_tools_tests gb_audio_buffer_tests gb_cartridge_tools_tests gb_feature_pack_tests gb_loader_fuzz_tests gb_miniz gb_lua gb_accuracy_tests gb_accuracy gb_achievements gb_achievement_tests)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
            if(NOT target STREQUAL "gb_miniz" AND NOT target STREQUAL "gb_lua")
                target_link_options(${target} PRIVATE /STACK:8388608)
            endif()
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-misleading-indentation)
            if(MINGW AND NOT target STREQUAL "gb_miniz" AND NOT target STREQUAL "gb_lua")
                target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
            endif()
            if(GB_ENABLE_SANITIZERS)
                target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
                target_link_options(${target} PRIVATE -fsanitize=address,undefined)
            endif()
        endif()
    endif()
endforeach()
