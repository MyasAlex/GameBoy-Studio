cmake_minimum_required(VERSION 3.20)
project(GameBoyStudio VERSION 0.6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_executable(gb_headless src/headless.cpp)
if(BUILD_TESTING)
    add_executable(gb_core_tests tests/core_tests.cpp)
    add_test(NAME core COMMAND gb_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb")
endif()
foreach(target gb_headless gb_core_tests)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-misleading-indentation)
        endif()
    endif()
endforeach()
if(WIN32)
    add_executable(GBStudio WIN32 src/windows_main.cpp)
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT GBStudio)
    target_link_libraries(GBStudio PRIVATE user32 gdi32 comdlg32 shell32 winmm)
    target_compile_definitions(GBStudio PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
        target_compile_options(GBStudio PRIVATE /W3 /EHsc /utf-8)
    else()
        target_compile_options(GBStudio PRIVATE -Wall -Wextra -Wno-misleading-indentation -Wno-unknown-pragmas)
        if(MINGW)
            target_link_options(GBStudio PRIVATE -municode -static -static-libgcc -static-libstdc++)
        endif()
    endif()
    add_custom_command(TARGET GBStudio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/demo.gb" "$<TARGET_FILE_DIR:GBStudio>/demo.gb")
    install(TARGETS GBStudio RUNTIME DESTINATION .)
    install(FILES demo.gb README_RU.md COMPATIBILITY.md DESTINATION .)
endif()
